      - name: Audit dependencies, licenses, secrets and configuration
        id: trivy
        run: |
          umask 077
          mkdir -p "$RUNNER_TEMP/security-private" "$RUNNER_TEMP/security-public"
          set +e
          docker run --rm \
            --user "$(id -u):$(id -g)" \
            -v "$GITHUB_WORKSPACE:/repo:ro" \
            -v "$RUNNER_TEMP/security-private:/evidence" \
            aquasec/trivy:0.70.0@sha256:be1190afcb28352bfddc4ddeb71470835d16462af68d310f9f4bca710961a41e \
            --cache-dir /evidence/cache \
            fs --scanners vuln,misconfig,secret,license \
            --severity HIGH,CRITICAL --exit-code 1 --no-progress \
            --format json --output /evidence/security.json /repo \
            >"$RUNNER_TEMP/security-private/scanner.log" 2>&1
          scan_exit=$?
          set -e
          printf '%s\n' "$scan_exit" >"$RUNNER_TEMP/security-private/exit-code"
          echo "exit_code=$scan_exit" >> "$GITHUB_OUTPUT"
      - name: Sanitize security evidence
        id: sanitize
        if: ${{ always() }}
        run: |
          python3 scripts/security_evidence.py \
            --raw "$RUNNER_TEMP/security-private/security.json" \
            --status-file "$RUNNER_TEMP/security-private/exit-code" \
            --db-metadata "$RUNNER_TEMP/security-private/cache/db/metadata.json" \
            --output "$RUNNER_TEMP/security-public/summary.json"
      - name: Upload public security summary
        id: evidence-upload
        if: ${{ always() && steps.sanitize.outcome == 'success' }}
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: security-summary-${{ github.sha }}-${{ github.run_attempt }}
          path: ${{ runner.temp }}/security-public/summary.json
          if-no-files-found: error
          retention-days: 7
      - name: Enforce security result
        if: ${{ always() }}
        env:
          SCAN_OUTCOME: ${{ steps.trivy.outcome }}
          SCAN_EXIT_CODE: ${{ steps.trivy.outputs.exit_code }}
          SANITIZE_OUTCOME: ${{ steps.sanitize.outcome }}
          UPLOAD_OUTCOME: ${{ steps.evidence-upload.outcome }}
        run: |
          test "$SCAN_OUTCOME" = success
          test "$SCAN_EXIT_CODE" = 0
          test "$SANITIZE_OUTCOME" = success
          test "$UPLOAD_OUTCOME" = success
          python3 scripts/security_evidence.py --check "$RUNNER_TEMP/security-public/summary.json"
      - name: Remove private scanner evidence
        if: ${{ always() }}
        run: rm -rf -- "${RUNNER_TEMP:?}/security-private"
